# 本地会议录音软件 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个只在 Alan 的 Mac 本地运行的原生菜单栏应用，通过全局快捷键立即录制全部系统声音和当前麦克风，并按日期保存为可恢复的单个 M4A 文件。

**Architecture:** Swift Package 包含共享核心、菜单栏应用和技术验证工具。AppKit 管理状态栏与生命周期，ScreenCaptureKit 输出系统声音与麦克风样本，AVFoundation 统一格式、按时间戳混音并可恢复写入；UI 只消费录音状态机，外部副作用都通过协议注入。

**Tech Stack:** Swift 6.3、AppKit、SwiftUI、ScreenCaptureKit、AVFoundation、CoreMedia、CoreAudio、Carbon、ServiceManagement、UserNotifications、XCTest、Swift Package Manager。

## Global Constraints

- 当前验证设备：Apple Silicon、macOS 26.5.1；Xcode 26.6、Swift 6.3.3。
- Swift Package 最低 macOS 15；所有结果必须在当前 macOS 26.5.1 实测。
- 不使用 BlackHole、Electron、Tauri、Web 运行时或第三方 Swift 包。
- 不联网、不上传、不做账号、分析或遥测。
- 同时录制全部系统声音和当前麦克风，最终混合成一个音轨，不保存屏幕画面。
- 默认快捷键 `Control + Option + R`，允许修改；不申请辅助功能或输入监控权限。
- 默认根目录 `/Users/alan/Documents/快速本地录音软件/录音文件/`，允许修改。
- 文件为 `YYYY-MM-DD/会议录音-YYYY-MM-DD-HH-mm-ss.m4a`；跨午夜不拆分，重名追加序号，绝不覆盖。
- AAC、48 kHz、立体声、128 kbps M4A；目标每小时低于 70 MB。
- 单次无固定时长上限；低于 1 GB 可用空间时拒绝开始。
- Alan UI：冷灰三层、1px 边框、5–8px 圆角、13px 主体字；红色只表示录音，低饱和棕色表示异常。
- 用户可见文档、验证记录、构建产物和录音名包含 `YYYY-MM-DD`；Swift 源文件按职责命名。
- 构建命令显式使用 `DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer`，不修改全局 `xcode-select`。
- 不提交 `.superpowers/`、`.DS_Store`、`.build/`、`dist/`、录音文件或本地 app。

```swift
public struct ActiveRecording: Equatable, Sendable {
    public let startedAt: Date
    public let workingURL: URL
    public init(startedAt: Date, workingURL: URL) {
        self.startedAt = startedAt; self.workingURL = workingURL
    }
}

public struct SavedRecording: Equatable, Sendable {
    public let startedAt: Date
    public let duration: TimeInterval
    public let fileURL: URL
    public let recovered: Bool
    public init(startedAt: Date, duration: TimeInterval, fileURL: URL, recovered: Bool) {
        self.startedAt = startedAt; self.duration = duration; self.fileURL = fileURL; self.recovered = recovered
    }
}

public struct RecordingFailure: Error, Equatable, Sendable {
    public enum Code: String, Sendable { case permission, storage, capture, microphone, write, finalize, hotkey }
    public let code: Code
    public let message: String
    public init(code: Code, message: String) { self.code = code; self.message = message }
}

public enum RecordingPhase: Equatable, Sendable {
    case idle, preparing
    case recording(ActiveRecording, warning: RecordingFailure?)
    case stopping(ActiveRecording)
    case failed(RecordingFailure)
}

public struct AudioLevels: Equatable, Sendable {
    public let system: Float
    public let microphone: Float
    public static let silent = AudioLevels(system: 0, microphone: 0)
}

public enum RecordingSessionEvent: Equatable, Sendable {
    case levels(AudioLevels)
    case warning(RecordingFailure?)
    case failed(RecordingFailure)
}

public protocol RecordingSessionManaging: Sendable {
    func events() async -> AsyncStream<RecordingSessionEvent>
    func start(at date: Date) async throws -> ActiveRecording
    func stop() async throws -> SavedRecording
}

@MainActor
public final class RecordingCoordinator: ObservableObject {
    @Published public private(set) var phase: RecordingPhase = .idle
    @Published public private(set) var audioLevels: AudioLevels = .silent
    private let session: any RecordingSessionManaging
    private let now: @Sendable () -> Date
    private var eventTask: Task<Void, Never>?

    public init(session: any RecordingSessionManaging, now: @escaping @Sendable () -> Date = Date.init) {
        self.session = session; self.now = now
        eventTask = Task { [weak self, session] in
            let events = await session.events()
            for await event in events { await self?.consume(event) }
        }
    }

    public func toggleRecording() async {
        switch phase {
        case .idle, .failed:
            phase = .preparing
            do { phase = .recording(try await session.start(at: now()), warning: nil) }
            catch { phase = .failed(Self.failure(from: error)) }
        case let .recording(active, _):
            phase = .stopping(active)
            do { _ = try await session.stop(); phase = .idle }
            catch { phase = .failed(Self.failure(from: error)) }
        case .preparing, .stopping:
            return
        }
    }

    private static func failure(from error: Error) -> RecordingFailure {
        if let failure = error as? RecordingFailure { return failure }
        return RecordingFailure(code: .capture, message: error.localizedDescription)
    }

    private func consume(_ event: RecordingSessionEvent) {
        switch event {
        case let .levels(levels): audioLevels = levels
        case let .warning(warning):
            if case let .recording(active, _) = phase { phase = .recording(active, warning: warning) }
        case let .failed(failure): phase = .failed(failure)
        }
    }
}
```

`RecordingModels.swift` 导入 Foundation；`RecordingCoordinator.swift` 导入 Combine 和 Foundation。测试文件内定义 `@MainActor final class SessionManagerSpy: RecordingSessionManaging`，用 `AsyncStream.makeStream()` 提供 events；`start` 返回固定 `ActiveRecording`、`stop` 返回固定 `SavedRecording`，并分别累加调用次数。增加测试向 stream yield `.warning` 和 `.levels`，验证 coordinator 更新录音 warning 和双音源值。

`RecordingCoordinator` 只允许 `idle/failed → preparing → recording → stopping → idle`；`preparing/stopping` 时忽略 toggle；未知错误转换为 `.capture`，已有 `RecordingFailure` 原样保留。

```swift
public struct AppSettings: Codable, Equatable, Sendable {
    public var recordingRoot: URL
    public var hotkey: HotkeyDescriptor
    public var launchAtLogin: Bool
}

public struct RecordingPaths: Equatable, Sendable {
    public let startedAt: Date
    public let directoryURL: URL
    public let workingURL: URL
    public let finalURL: URL
    public let recoveredURL: URL
}

public actor RecordingStore {
    public static let minimumFreeBytes: Int64 = 1_000_000_000
    public func prepare(startedAt: Date) throws -> RecordingPaths {
        guard try availableCapacity() >= Self.minimumFreeBytes else {
            throw RecordingFailure(code: .storage, message: "可用空间不足 1 GB，未开始录音。")
        }
        let folder = format(startedAt, "yyyy-MM-dd")
        let stamp = format(startedAt, "yyyy-MM-dd-HH-mm-ss")
        let directory = root.appendingPathComponent(folder, isDirectory: true)
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        let final = uniqueURL(directory: directory, base: "会议录音-\(stamp)", extension: "m4a")
        return RecordingPaths(startedAt: startedAt, directoryURL: directory,
            workingURL: directory.appendingPathComponent(".\(final.deletingPathExtension().lastPathComponent).inprogress.mov"),
            finalURL: final,
            recoveredURL: directory.appendingPathComponent("\(final.deletingPathExtension().lastPathComponent)-未完整恢复.m4a"))
    }
}
```

formatter 使用 `en_US_POSIX` 和注入时区；`uniqueURL` 检查原名、`-2`、`-3`。`AppSettingsStore` 注入独立 `UserDefaults`，首次值精确为默认目录、默认快捷键、`launchAtLogin=true`；验证写入后重建回读。容量测试用 `do/catch` 断言 `999_999_999` bytes 返回 `.storage`，不用未定义的异步 XCTest helper。

`recoveredURL(for:)` 去掉开头的点和 `.inprogress.mov`，在同一日期目录生成 `原基础名-未完整恢复.m4a`，并同样执行唯一命名检查。

```swift
public enum CapturedAudioSource: Equatable, Sendable { case system, microphone }

public struct CapturedAudioSample: @unchecked Sendable {
    public let source: CapturedAudioSource
    public let buffer: CMSampleBuffer
    public let presentationTime: CMTime
}

public enum AudioCaptureEvent: @unchecked Sendable {
    case sample(CapturedAudioSample)
    case microphoneChanged(String)
    case warning(RecordingFailure)
    case stopped(RecordingFailure?)
}

public static func makeConfiguration(microphoneDeviceID: String?) -> SCStreamConfiguration {
    let config = SCStreamConfiguration()
    config.capturesAudio = true
    config.captureMicrophone = true
    config.microphoneCaptureDeviceID = microphoneDeviceID
    config.sampleRate = 48_000
    config.channelCount = 2
    config.excludesCurrentProcessAudio = true
    config.width = 2; config.height = 2
    config.minimumFrameInterval = CMTime(value: 1, timescale: 1)
    return config
}
```

两路使用独立串行队列；delegate 只校验 sample 并发送事件，不做文件 I/O。先不注册 `.screen` output；若真实 probe 证明无法启动，才注册 2×2、1 fps screen output 并立即丢弃帧，报告必须记录。

content filter 选择当前 display、排除本应用进程，但不排除任何其他应用，因此 Zoom、浏览器、腾讯会议、企业微信和系统通知都属于系统音频范围。用 CoreAudio 的 `kAudioHardwarePropertyDefaultInputDevice` listener 观察默认麦克风变化；listener 只触发 `updateDefaultMicrophone()`，失败产生 `.microphone` warning，不停止系统音频。

```swift
public actor LiveRecordingSessionManager: RecordingSessionManaging {
    private enum State { case idle, recording(Context), stopping }
    private var state: State = .idle

    public func start(at date: Date) async throws -> ActiveRecording {
        guard case .idle = state else { throw RecordingFailure(code: .capture, message: "已有录音正在进行。") }
        let permission = await permissions.currentStatus()
        guard permission.isGranted else { throw RecordingFailure(code: .permission, message: permission.userMessage) }
        let paths = try await store.prepare(startedAt: date)
        let writer = try writerFactory.makeWriter(outputURL: paths.workingURL)
        sleep.begin()
        do {
            try await writer.start()
            try await capture.start { [weak self] event in await self?.handle(event) }
            let active = ActiveRecording(startedAt: date, workingURL: paths.workingURL)
            state = .recording(Context(active: active, paths: paths, writer: writer))
            return active
        } catch {
            await capture.stop(); await writer.abort(); sleep.end(); state = .idle
            throw error
        }
    }
}
```

sample → convert → meter RMS → mixer → writer。麦克风失败只叠加 warning；系统捕获/写入失败触发一次停止。固定停止顺序同测试；失败保留 working file。

```swift
public struct RecoveryResult: Equatable, Sendable {
    public enum Outcome: Equatable, Sendable { case recovered(URL), failed(URL, String) }
    public let outcome: Outcome
}

public actor RecoveryService {
    public func recoverInterruptedRecordings() async -> [RecoveryResult] {
        let files = (try? await store.listInterruptedRecordings()) ?? []
        var results: [RecoveryResult] = []
        for workingURL in files {
            do {
                let recoveredURL = await store.recoveredURL(for: workingURL)
                let output = try await finalizer.recover(workingURL: workingURL, recoveredURL: recoveredURL)
                results.append(.init(outcome: .recovered(output)))
            } catch {
                results.append(.init(outcome: .failed(workingURL, error.localizedDescription)))
            }
        }
        return results
    }
}
```
